import json import os from datetime import datetime, timedelta # ====================== 【核心配置】 ====================== # 数据文件（自动保存，下次打开还在） DATA_FILE = "word_data.json" # 艾宾浩斯复习曲线（按天，不背单词同款） REVIEW_CURVE = [1, 2, 4, 7, 15, 30] # 每轮学习/复习单词数量 BATCH_SIZE = 10 # ========================================================= # ====================== 【数据加载/保存】 ====================== def load_data(): if not os.path.exists(DATA_FILE): return { "words": [], "study_record": {}, "today": str(datetime.now().date()), "continuous_days": 0, "total_money": 0.0 } with open(DATA_FILE, "r", encoding="utf-8") as f: return json.load(f) def save_data(data): with open(DATA_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) # 自动更新日期（今天进是今天，明天进是明天） def update_date(data): today = str(datetime.now().date()) if data["today"] != today: data["today"] = today # 连续学习天数自动计算 yesterday = str((datetime.now() - timedelta(days=1)).date()) if yesterday in data["study_record"]: data["continuous_days"] += 1 else: data["continuous_days"] = 1 save_data(data) # ====================== 【单词核心功能】 ====================== # 初始化单词库（N5示例） def init_words(data): if not data["words"]: default_words = [ {"word": "私", "kana": "わたし", "sound": "watashi", "example": "私は学生です", "collocation": "私たち", "status": "new", "review_count": 0, "next_review": "", "add_time": str(datetime.now().date())}, {"word": "本", "kana": "ほん", "sound": "hon", "example": "本を買う", "collocation": "本屋", "status": "new", "review_count": 0, "next_review": "", "add_time": str(datetime.now().date())}, {"word": "学校", "kana": "がっこう", "sound": "gakkou", "example": "学校へ行く", "collocation": "学校生活", "status": "new", "review_count": 0, "next_review": "", "add_time": str(datetime.now().date())} ] data["words"] = default_words * 4 # 凑够足够数量 save_data(data) # 获取今日需要学习/复习的单词 def get_today_words(data, mode="review"): today = str(datetime.now().date()) result = [] for w in data["words"]: if mode == "new" and w["status"] == "new": result.append(w) if mode == "review" and w["next_review"] == today: result.append(w) return result[:BATCH_SIZE] # 更新复习计划 def update_review(word, is_forget=False): if is_forget: word["status"] = "new" word["review_count"] = 0 word["next_review"] = "" return word["review_count"] += 1 if word["review_count"] >= len(REVIEW_CURVE): word["status"] "mastered" word["next_review"] = "" else: days = REVIEW_CURVE[word["review_count"]-1] next_day = str((datetime.now() + timedelta(days=days)).date()) word["next_review"] = next_day word["status"] = "studying" # ====================== 【页面1：首页】 ====================== def home_page(data): update_date(data) while True: print("\n" + "="*30) print(" 🏠 首页") print("="*30) print("1. 📖 学习新单词") print("2. 🔁 复习已学单词") print("3. 📊 学习进程概览") print("0. 🚪 退出") choice = input("请选择：") if choice == "1": study_page(data, "new") elif choice == "2": study_page(data, "review") elif choice == "3": overview_page(data) elif choice == "0": print("再见！") break # ====================== 【页面2：学习/复习页面】 ====================== def study_page(data, mode): today = str(datetime.now().date()) words = get_today_words(data, mode) if not words: print("✅ 今日无任务！") return learned = 0 for word in words: print("\n" + "-"*40) print(f"单词：{word['word']}") print(f"假名：{word['kana']}") print(f"发音：{word['sound']}") print(f"例句：{word['example']}") print(f"搭配：{word['collocation']}") print(f"复习次数：{word['review_count']}") if word['next_review']: print(f"下次复习：{word['next_review']}") print("-"*40) # 记录学习 if today not in data["study_record"]: data["study_record"][today] = {"count": 0, "time": 0} data["study_record"][today]["count"] += 1 while True: opt = input("1. 下一个 2. 记错了：") if opt == "1": update_review(word, is_forget=False) learned += 1 break elif opt == "2": update_review(word, is_forget=True) break # 攒钱激励：每10个 = 1元 money = learned / 10 data["total_money"] += money save_data(data) # 进入拼写/小结 spelling_page(data, words) # ====================== 【页面3：强化拼写 + 小结】 ====================== def spelling_page(data, words): print("\n🎯 学习完成！选择下一步：") print("1. 强化拼写") print("2. 小结（跳过）") choice = input("请选择：") if choice == "1": print("\n✍️ 开始强化拼写") wrong_words = words.copy() while wrong_words: for w in wrong_words[:]: ans = input(f"请拼写：{w['kana']} → ") if ans == w["word"]: wrong_words.remove(w) print(f"✅ 正确！") else: print(f"❌ 错误，重新来！") # 展示结果列表 print("\n📋 本次学习单词") for w in words: print(f"{w['word']} | 下次复习：{w['next_review'] or '已掌握'}") input("\n按回车返回首页") # ====================== 【页面4：学习进程概览】 ====================== def overview_page(data): today = str(datetime.now().date()) today_data = data["study_record"].get(today, {"count": 0, "time": 0}) total_count = sum([v["count"] for v in data["study_record"].values()]) print("\n" + "="*40) print(" 📊 学习进程概览") print("="*40) # 顶部 print(f"📚 当前学习：N5单词") # 中部数据 print(f"\n📈 我的数据") print(f"今日学习/复习：{today_data['count']} 词") print(f"累计学习：{total_count} 词") print(f"今日时长：{today_data['time']} 分钟") print(f"累计时长：sum([v['time'] for v in data['study_record'].values()]) 分钟") # 底部日历 + 激励 print(f"\n📅 学习日历（点击查看完整）") print(f"❌未学 ✅10个 😊10-20个 👍≥20个") print(f"\n🔥 连续学习：{data['continuous_days']} 天 💰 累计攒钱：{data['total_money']:.1f} 元") input("\n按回车返回") # ====================== 【启动】 ====================== if __name__ == "__main__": data = load_data() update_date(data) init_words(data) home_page(data)
